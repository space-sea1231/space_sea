###### 1.[P3379](https://www.luogu.com.cn/problem/P3379)【模板】最近公共祖先（LCA）

这题是昨天没写完今天接着续写完的，只用了20分钟，不错，就当开胃菜了

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N=5e5+10;
int n, m, s, head[N], dep[N], dp[N][21], cnt;
struct node{
    int to, next;
}e[N*2];
void add(int u, int v){
    e[++cnt].to=v;
    e[cnt].next=head[u];
    head[u]=cnt;
}
void serch(int now, int fa){
    for (int i=head[now]; i; i=e[i].next){
        int v=e[i].to;
        if (v==fa){
            continue;
        }
        dep[v]=dep[now]+1;
        dp[v][0]=now;
        for (int j=1; j<=20; j++){
            dp[v][j]=dp[dp[v][j-1]][j-1];
        }
        serch(v, now);
    }
}
int lca(int x, int y){
    if (dep[x]<dep[y]){
        int z=x;
        x=y;
        y=z;
    }
    for (int i=20; i>=0; i--){
        if (dep[dp[x][i]]>=dep[y]){
            x=dp[x][i];
        }
    }
    if (x==y){
        return x;
    }
    for(int i=20;i>=0;i--){
        if(dp[x][i]!=dp[y][i]){
            x=dp[x][i];
            y=dp[y][i];
        }
    }
    return dp[x][0];
}
int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> m >> s;
    int u, v;
    for (int i=1; i<n; i++){
        cin >> u >> v;
        add(u, v);
        add(v, u);
    }
    // cerr << "Debug:in\n";
    dep[s]=1;
    serch(s, 0);
    // cerr << "Debug:out\n";
    int x, y;
    for (int i=1; i<=m; i++){
        cin >> x >> y;
        // cerr << "Debug2:in " << i << endl;
        printf("%d\n", lca(x, y));
    }
    
    return 0;
}
```

###### 3.[P3388](https://www.luogu.com.cn/problem/P3388)【模板】割点（割顶）

在这题中第一次运用了Tarjan进行树的遍历，有迈进了一步（ovo）

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N=2e4+10;
const int M=1e5+10;
int head[N], cnt, n, m;//建边系列
int dfn[N], low[N], tot;//tarjan系列 
bool cut[N];
struct node{
    int to, next;
}e[M*2];
void add(int u, int v){
    e[++cnt].to=v;
    e[cnt].next=head[u];
    head[u]=cnt;
}
void tarjan(int u, int root){
    int child=0;
    dfn[u]=low[u]=++tot;
    for (int i=head[u]; i; i=e[i].next){
        int v=e[i].to;
        if (!dfn[v]){
            tarjan(v, root);
            low[u]=min(low[u], low[v]);
            if (low[v]>=dfn[u]&&u!=root){
                cut[u]=1;
            }
            if (u==root){
                child++;
            }
        }
        low[u]=min(low[u], dfn[v]);
    }
    if (child>=2&&u==root){
        cut[u]=1;
    }
}
int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> m;
    int u, v;
    for (int i=1; i<=m; i++){
        cin >> u >> v;
        add(u, v);
        add(v, u);
    }
    for (int i=1; i<=n; i++){
        if (dfn[i]){
            continue;
        }
        tarjan(i, i);
    }
    int sum=0;
    for (int i=1; i<=n; i++){
        if (cut[i]){
            sum++;
        }
    }
    printf("%d\n", sum);
    for (int i=1; i<=n; i++){
        if (cut[i]){
            printf("%d ", i);
        }
    }
    return 0;
}
```

```cpp
#include <bits/stdc++.h>
#define int long long
using namespace std;
const int N=2e6+10;
int n, m, a[N];
struct node{
    int l, r, sum, vis;
}e[N*2];
void build(int x, int l, int r){
    int mid=(l+r)>>1;
    e[x].l=l;
    e[x].r=r;
    if (l==r){
        e[x].sum=a[l];
        return;
    }
    build(x*2, l, mid);
    build(x*2+1, mid+1, r);
    e[x].sum=e[x*2].sum+e[x*2+1].sum;
}
void down(int x){
    e[x*2].vis+=e[x].vis;
    e[x*2+1].vis+=e[x].vis;
    e[x*2].sum+=(e[x*2].r-e[x*2].l+1)*e[x].vis;
    e[x*2+1].sum+=(e[x*2+1].r-e[x*2+1].l+1)*e[x].vis;
    e[x].vis=0;
}
void add(int x, int l, int r, int w){
    if (e[x].l>r||e[x].r<l){
        return;
    }
    if (e[x].l>=l&&e[x].r<=r){
        e[x].vis+=w;
        e[x].sum+=(e[x].r-e[x].l+1)*w;
        return;
    }
    if (e[x].vis){
        down(x);
    }
    add(x*2, l, r, w);
    add(x*2+1, l, r, w);
    e[x].sum=e[x*2].sum+e[x*2+1].sum;
}
int sum(int x, int l, int r){
    if (e[x].l>r||e[x].r<l){
        return 0;
    }
    if (e[x].l>=l&&e[x].r<=r){
        return e[x].sum;
    }
    if (e[x].vis){
        down(x);
    }
    return sum(x*2, l, r)+sum(x*2+1, l, r);
}
```

```cpp
#define int long long
using namespace std;
const int N=2e5+10;
int n, m, Mod, a[N];
struct node{

}e[N*2];
void build(int x, int l, int r){
    int mid=(l+r)>>1;
    e[x].l=l;
    e[x].r=r;
    e[x].vis_m=1;
    if (l==r){
        e[x].sum=a[l]%Mod;
        return;
    }
    build(x*2, l, mid);
    build(x*2+1, mid+1, r);
    e[x].sum=(e[x*2].sum+e[x*2+1].sum)%Mod;
}
void down(int x){
    e[x*2].sum=(e[x*2].sum*e[x].vis_m+(e[x*2].r-e[x*2].l+1)*e[x].vis_a)%Mod;
    e[x*2+1].sum=(e[x*2+1].sum*e[x].vis_m+(e[x*2+1].r-e[x*2+1].l+1)*e[x].vis_a)%Mod;
    e[x*2].vis_a=(e[x*2].vis_a*e[x].vis_m+e[x].vis_a)%Mod;
    e[x*2+1].vis_a=(e[x*2+1].vis_a*e[x].vis_m+e[x].vis_a)%Mod;
    e[x*2].vis_m=(e[x*2].vis_m*e[x].vis_m)%Mod;
    e[x*2+1].vis_m=(e[x*2+1].vis_m*e[x].vis_m)%Mod;
    e[x].vis_a=0;
    e[x].vis_m=1;
}
void mul(int x, int l, int r, int w){
    int mid=(e[x].l+e[x].r)>>1;
    if (e[x].l>=l&&e[x].r<=r){
        e[x].vis_a=e[x].vis_a*w%Mod;
        e[x].vis_m=e[x].vis_m*w%Mod;
        e[x].sum=e[x].sum*w%Mod;
        return;
    }
    down(x);
    if (l<=mid){
        mul(x*2, l, r, w);
    }
    if (mid<r){
        mul(x*2+1, l, r, w);
    }
    e[x].sum=(e[x*2].sum+e[x*2+1].sum)%Mod;
}
void add(int x, int l, int r, int w){
    int mid=(e[x].l+e[x].r)>>1;
    if (e[x].l>=l&&e[x].r<=r){

        e[x].sum=(e[x].sum+(e[x].r-e[x].l+1)*w)%Mod;

        add(x*2, l, r, w);
    }
    if (mid<r){
        add(x*2+1, l, r, w);
    }

}

    if (e[x].l>=l&&e[x].r<=r){
        return e[x].sum;
    }
    down(x);
    int ans=0;
    if (l<=mid){
        ans=(ans+sum(x*2, l, r))%Mod;
    }
    if (mid<r){
        ans=(ans+sum(x*2+1, l, r))%Mod;

    return ans;
}
```
