##### T1

###### 得分：100分

###### 思路

题目很简单，就是消消乐，栈的模板题。先按坐标排一遍序（结构体存数据），扫一遍数组，将向右滚动的小球全部丢到栈中，碰到一个向左滚动的就弹出一个，把它们两抵消掉，结果就是它们的坐标差的绝对值。如果栈为空但是碰到向左滚动的小球，那这个球就会一直滚动下去（-1的情况）。同理，如果最后栈中还剩下向右滚动的小球，那就一个一个弹出并标记为-1

##### [**P8436**​](https://www.luogu.com.cn/problem/P8436)【模板】边双连通分量

**边双连通分量：**`**一张连通图的极大边双连通子图**`

**边双联通图：**`**图中任意两个点都至少两条不同路径可以到达的图**`

这题又是Tarjan的模板题，但是有个前置芝士：割边/桥

**割边/桥：** **假如删去这条边后图不连通，那么称这条边为割边**

在此基础上略微修改就可以得到边双联通分量

```cpp
#include<bits/stdc++.h>
using namespace std;
const int N=5e5+10;
const int M=2e6+10;
int head[N], cnt=1, n, m;//建边系列
int dfn[N], low[N], c[N], num, tot;//Tarjan系列
bool bridge[M*2];
vector<int> q[N];
struct node{
    int to, next;
}e[M*2];
void add(int u, int v){
    e[++cnt].to=v;
    e[cnt].next=head[u];
    head[u]=cnt;
}
void Tarjan(int u, int in_edge){
    dfn[u]=low[u]=++num;
    for (int i=head[u]; i; i=e[i].next){
        int v=e[i].to;
        if (!dfn[v]){//树枝边
            Tarjan(v, i);
            low[u]=min(low[u], low[v]);
            if (low[v]>dfn[u]){
                bridge[i]=bridge[i^1]=1;
                // printf("Debug2: %d %d\n", u, v);
            }
        }else if(i!=(in_edge^1)){//反祖边
            low[u]=min(low[u], dfn[v]);
        }
    }
}
void dfs(int u){
    // printf("Debug1: %d %d\n", tot, u);
    q[tot].push_back(u);
    c[u]=tot;
    int v;
    for (int i=head[u]; i; i=e[i].next){
        v=e[i].to;
        if (c[v]||bridge[i]){
            continue;
        }
        dfs(v);
    }
}
int main(){
	ios::sync_with_stdio(0);
	cin.tie(0);
    cin >> n >> m;
    int u, v;
    for (int i=1; i<=m; i++){
        cin >> u >> v;
        if (u==v){
            continue;
        }
        add(u, v);
        add(v, u);
    }
    for (int i=1; i<=n; i++){
        if (dfn[i]){
            continue;
        }
        Tarjan(i, 0);
    }
    for (int i=1; i<=n; i++){
        if (c[i]){
            continue;
        }
        tot++;
        dfs(i);
    }
    printf("%d\n", tot);
    int siz;
    for (int i=1; i<=tot; i++){
        siz=q[i].size();
        printf("%d ", siz);
        for (int j=0; j<siz; j++){
            printf("%d ", q[i][j]);
        }
        printf("\n");
    }
    return 0;
}
```

##### [**P8435**](https://www.luogu.com.cn/problem/P8435)​【模板】点双连通分量

**点双连通分量：**`**对于一个无向图中的极大点双连通的子图，我们称这个子图为点双连通分量**`

**点双联通图：**`**假如仅仅对于该图而言其中不包含割点，那么称这个图是点双连通的**`

这题双是Tarjan的模板题，但是有个前置芝士：割点

**割点：**`**对于一个连通图中的点，假如删去这个点以及与所有与该点相连的边之后图不连通，那么称该点为该图的割点**`

此基础上略微修改就可以得到点双联通分量

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N=5e5+10;
const int M=2e6+10;
int n, m, head[N], cnt;//建边系列
int dfn[N], low[N], sum[N], tot, num, top;//Tarjan系列
/*
dfn[u] 节点u的时间戳（访问顺序）
low[u] 节点u的回溯值（最多通过一条非树边所能到达时间戳最小的节点时间戳）
sum[] 手写栈 
tot 点双连通分量的个数
num 时间戳编号
top 栈顶编号
*/
vector<int> q[N];
struct node{
    int to, next;
}e[M*2];
void add(int u, int v){
    e[++cnt].to=v;
    e[cnt].next=head[u];
    head[u]=cnt;
}
void Tarjan(int u, int root){
    dfn[u]=low[u]=++num;
    sum[++top]=u;
    if (u==root&&head[u]==0){//判断孤立点
        q[++tot].push_back(u);
        return;
    }
    int v;
    for (int i=head[u]; i; i=e[i].next){
        v=e[i].to;
        if (!dfn[v]){//目标节点未被访问过
            Tarjan(v, root);
            low[u]=min(low[u], low[v]);//更新时间戳
            if (low[v]>=dfn[u]){//子树无法访问到u的祖先节点
                tot++;
                int now;
                do{//保证弹出当前节点
                    now=sum[top--];
                    q[tot].push_back(now);
                }while(now!=v);
                q[tot].push_back(u);//将当前节点一起弹进去
            }
        }
        low[u]=min(low[u], dfn[v]);
    }
}
int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);//读入优化
    cin >> n >> m;
    int u, v;
    for (int i=1; i<=m; i++){
        cin >> u >> v;
        if (u==v){//判断自环
            continue;
        }
        add(u, v);
        add(v, u);
    }
    for (int i=1; i<=n; i++){
        if (dfn[i]){//访问过了
            continue;
        }
        Tarjan(i, i);
    }
    printf("%d\n", tot);
    int siz;
    for (int i=1; i<=tot; i++){
        siz=q[i].size();
        printf("%d ", siz);
        for (int j=0; j<siz; j++){
            printf("%d ", q[i][j]);
        }
        printf("\n");
    }
    return 0;
}
```

##### [**P3388**​](https://www.luogu.com.cn/problem/P3388)【模板】割点（割顶）

*前置芝士后面做，绝了*

这题叒是Tarjan的模板题，但是有个前置芝士：Tarjan（你逗我玩呢？）

此基础上略微修改就可以得到割点

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N=2e4+10;
const int M=1e5+10;
int head[N], cnt, n, m;//建边系列
int dfn[N], low[N], tot, num;//Tarjan系列
bool cut[N];
struct node{
    int to, next;
}e[M*2];
void add(int u, int v){
    e[++cnt].to=v;
    e[cnt].next=head[u];
    head[u]=cnt;
}
void Tarjan(int u, int root){
    dfn[u]=low[u]=++num;
    int v, child=0;
    for (int i=head[u]; i; i=e[i].next){
        v=e[i].to;
        if (!dfn[v]){
            Tarjan(v, root);
            low[u]=min(low[u], low[v]);
            if (low[v]>=dfn[u]){
                child++;
                if (child>1||u!=root){
                    cut[u]=1;
                }
            }
        }
        low[u]=min(low[u], dfn[v]);
    }
}
int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> m;
    int u, v;
    for (int i=1; i<=m; i++){
        cin >> u >> v;
        add(u, v);
        add(v, u);
    }
    for (int i=1; i<=n; i++){
        if (dfn[i]){
            continue;
        }
        Tarjan(i, i);
    }
    int cntt=0;
    for (int i=1; i<=n; i++){
        if (cut[i]){
            cntt++;
        }   
    }
    printf("%d\n", cntt);
    for (int i=1; i<=n; i++){
        if (cut[i]){
            printf("%d ", i);
        }
    }
    return 0;
}
```
