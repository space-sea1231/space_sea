###### [T525503](https://www.luogu.com.cn/problem/T525503)【ZZB‘s Team's Test 3 A】按位与取累了（and）

得分为0分qwq.....

先将每个数进行二进制拆分，用*vector*存起来，从高位往低位枚举，如果有至少两个数当前这一位为**TRUE**，那就现累加这一位在十进制下的值，随后的枚举就只在这几个数中寻找，以此类推......

```cpp
#include <bits/stdc++.h>
#define int long long
using namespace std;
const int N=5e6+10;
const int bit=32;
int n;
bool cnt[N][bit], vis[N];
vector<int> q[bit];
signed main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n;
    int a, now;
    for (int i=1; i<=n; i++){
        cin >> a;
        for (int j=31; j>=0; j--){
            //1010 
            //1001 
            //1010 
            //1000
            now=(1LL<<j);
            // cerr << now << endl;
            if (a>=now){
                // printf("%lld %lld %lld\n", i, a, now);
                a-=now;
                cnt[i][j]=1;
                q[j].push_back(i);
            }
        }
    }
    int siz, sum=0;
    for (int i=31; i>=0; i--){
        if (q[i].size()>=2){
            now=i;
            break;
        }
    }
    // printf("Debug1: %lld\n", now);
    siz=q[now].size();
    for (int i=0; i<siz; i++){
        vis[q[now][i]]=1;
    }
    for (int i=now; i>=0; i--){
        if (q[i].size()>=2){
            sum+=(1<<i);
            // printf("%lld %lld\n", i, sum);
        }else{
            continue;
        }
        siz=q[i].size();
        for (int j=1; j<=n; j++){
            if (vis[j]==1&&cnt[j][now]==0){
                vis[j]=0;
                for (int k=i; k>=0; k--){
                    if (cnt[j][k]){
                        q[k].pop_back();
                    }
                }
            }
        }
    }
    printf("%lld", sum);

    return 0;
}
```

###### [T525521](https://www.luogu.com.cn/problem/T525521)【ZZB‘s Team's Test 3 B】借到钱就跑了（money）

得分为40分...为什么?这题不就是一道dp的板子题吗?怎么只有40分?

`dp[i][0]`表示这一天1取家长的钱的最大值,转移方程很好求,直接由昨天取银行和同学的钱中的最大值加上今天从家长那~~讹借~~来的钱就行了,其他两个同样。最后的答案就是`max(dp[n][0], max(dp[n][1], dp[n][1]))`

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N=1e6+10;
int a[N], b[N], c[N], dp[N][3], n;
int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n;
    for (int i=1; i<=n; i++){
        cin >> a[i] >> b[i] >> c[i];
    }
    for (int i=1; i<=n; i++){
        dp[i][0]=max(dp[i-1][1]+a[i], dp[i-1][2]+a[i]);
        dp[i][1]=max(dp[i-1][0]+b[i], dp[i-1][2]+b[i]);
        dp[i][2]=max(dp[i-1][0]+c[i], dp[i-1][1]+c[i]);
    }
    printf("%d", max(dp[n][0], max(dp[n][1], dp[n][2])));

    return 0;
}
```

###### [T526585](https://www.luogu.com.cn/problem/T526585)【ZZB‘s Team's Test 3 C】被拒绝就哭了（cry）

得分：100分！

这一题是动态规划和单调队列的结合题，每次运用单调队列处理当前行的上一行，很容易的就可以进行动态转移了

```cpp
#include <bits/stdc++.h>
using namespace std;
const int N=8e3+10;
const int INF=1e9;
int n, m, k, t, tail, head, q[N], dp[N][N], a[N][N];
void init(int x){
    for (int i=1; i<=t; i++){
        while (dp[x][i]>dp[x][q[tail]]&&tail>=head){
            tail--;
        }
        q[++tail]=i; 
    }
}
void go(int x, int now){
	if (x+t<=m){
        while (dp[now][x+t]>dp[now][q[tail]]&&tail>=head){
            tail--;
        }
        q[++tail]=x+t;
    }
	while(q[head]+t<x){
        head++;
    }
}
int main(){
    ios::sync_with_stdio(0);
    cin.tie(0);
    cin >> n >> m >> k >> t;
    int x, y, w;
    for (int i=1; i<=k; i++){
        cin >> x >> y >> w;
        a[x][y]=w;
    }
    for (int i=1; i<=n; i++){
   	    dp[1][i]=a[1][i];
    }
    for (int i=2; i<=n; i++){
        head=1, tail=0;
        init(i-1);
        for (int j=1; j<=m; j++){
            go(j, i-1);
            dp[i][j]=dp[i-1][q[head]]+a[i][j];
        }
    }
    int maxn=-INF;
    for (int i=1; i<=m; i++){
        maxn=max(maxn, dp[n][i]);
    }
    printf("%d", maxn);

    return 0;
}
```
