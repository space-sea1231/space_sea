> 强连通分量（strongly connect component）scc

## 定义

强连通：在一个有向图中，若点x和点y能彼此到达，则称x和y是强连通的。

强连通分量（scc）：在一张有向图G中，G'是G的一个子图，G'内的点两两强连通，且G'不能再大，称子图G'是G的一个scc。

注意：

1.  scc要么是一个环，要么是一个单点。
2.  scc不一定是简单环。

使用场景：将一张带环的有向图转化为DAG（有向无环图 缩点）

## 模板

### P2863 [USACO06JAN] The Cow Prom S

#### 思路

1.  维护dfs序的时间戳dfn[x]和最小时间戳low[x]，low[x]表示x出发能到达的节点中最小的时间戳。
2.  将访问的点插入栈中，当找到关键点时，则栈顶到关键点的所有点属于同一个scc。

> 关键点：在一个scc中率先被访问到的点，它的low值=dfn值

#### 代码

```cpp
#include<bits/stdc++.h>
using namespace std;

const int N=5e4+5;
int n,m,dfn[N],low[N],tot[N],cnt,sum,ans;
bool instk[N];
vector<int>nbr[N];
stack<int>stk;

void Tarjan(int cur)
{
	stk.push(cur);
	instk[cur]=true;
	dfn[cur]=low[cur]=++cnt;
	for(int i=0;i<nbr[cur].size();i++)
	{
		int nxt=nbr[cur][i];
		if(dfn[nxt]==0)
		{
			Tarjan(nxt);
			low[cur]=min(low[cur],low[nxt]);
		}
		else if(instk[nxt]==true)
			low[cur]=min(low[cur],low[nxt]);
	}
	if(dfn[cur]==low[cur])
	{
		sum++;
		while(stk.top()!=cur)
		{
			int x=stk.top();
			stk.pop();
			instk[x]=false;
			tot[sum]++;
		}
		instk[cur]=false;
		tot[sum]++;
		stk.pop();
	}
	return ;
}

int main()
{
	cin>>n>>m;
	for(int i=1;i<=m;i++)
	{
		int x,y;
		cin>>x>>y;
		nbr[x].push_back(y);
	}
	for(int i=1;i<=n;i++)
		if(dfn[i]==0)
			Tarjan(i);
	for(int i=1;i<=sum;i++)
		if(tot[i]>1)
			ans++;
	cout<<ans;
	return 0;
}
```

## 练习

### P3916 图的遍历

#### 注意

本题在弹出时维护mxid[x]，在Tarjan结束时建新图，跑dfs或者拓扑排序即可。

#### 代码

```cpp
#include<bits/stdc++.h>
using namespace std;
#define int long long

const int N=1e5+5;
int n,m,x[N],y[N],dfn[N],low[N],scc[N],cnt,sum,ans,mxid[N],in[N];
bool instk[N],vis[N];
vector<int>nbr[N],newnbr[N];
stack<int>stk;

void Tarjan(int cur)
{
	stk.push(cur);
	instk[cur]=true;
	dfn[cur]=low[cur]=++cnt;
	for(int i=0;i<nbr[cur].size();i++)
	{
		int nxt=nbr[cur][i];
		if(dfn[nxt]==0)
		{
			Tarjan(nxt);
			low[cur]=min(low[cur],low[nxt]);
		}
		else if(instk[nxt]==true)
			low[cur]=min(low[cur],low[nxt]);
	}
	if(dfn[cur]==low[cur])
	{
		sum++;
		while(stk.top()!=cur)
		{
			int x=stk.top();
			stk.pop();
			instk[x]=false;
			scc[x]=sum;
			mxid[sum]=max(mxid[sum],x);
		}
		instk[cur]=false;
		scc[cur]=sum;
		mxid[sum]=max(mxid[sum],cur);
		stk.pop();
	}
	return ;
}

void dfs(int cur)
{
	vis[cur]=true;
	for(int i=0;i<newnbr[cur].size();i++)
	{
		int nxt=newnbr[cur][i];
		if(vis[nxt]==false)
			dfs(nxt);
		mxid[cur]=max(mxid[cur],mxid[nxt]);
	}
	return ;
}

signed main()
{
	memset(mxid,0x80,sizeof mxid);
	cin>>n>>m;
	for(int i=1;i<=m;i++)
	{
		cin>>x[i]>>y[i];
		nbr[x[i]].push_back(y[i]);
	}
	for(int i=1;i<=n;i++)
		if(dfn[i]==0)
			Tarjan(i);
	for(int i=1;i<=m;i++)
	{
		int X=scc[x[i]],Y=scc[y[i]];
		if(X!=Y)
		{
			newnbr[X].push_back(Y);
			in[Y]++;
		}
	}
	for(int i=1;i<=sum;i++)
		if(in[i]==0)
			newnbr[0].push_back(i);
	dfs(0);
	for(int i=1;i<=n;i++)
		cout<<mxid[scc[i]]<<" ";
	return 0;
}
```
