> 使用场景：在树上任意路径上，做点权或边权的修改。

前置知识：[LCA](https://memset.yuque.com/org-wiki-memset-fk7vsw/kgy4gn/lgnek9zdg0hrn2bc)。

## 关于点的差分

假设将节点x到y的所有点权增加$val$，等价于`diff[x]+=val,diff[fa[lca(x,y)]]-=val;diff[y]+=val,diff[fa[lca(x,y)]]-=val;`

当所有修改执行完毕，跑树形DP，将子节点的`diff[]`累加到父节点，得到原数组a[];

伪代码：

```cpp
void dfs(int cur,int fa)
{
    a[cur]=diff[cur];
    for(int i=0;i<nbr[cur].size();i++)
    {
        int nxt=nbr[cur][i];
        if(nxt!=fa)
        {
            dfs(nxt,fa);
            a[cur]+=a[nxt];
        }
    }
    return ;
}
```

## 例题

### P3128 [USACO15DEC] Max Flow P

#### 基础做法

1.  对于每条路径$x\to y$，先求出`lca(x,y)`。
2.  从$x$开始向上移动直到`lca(x,y)`，沿途点$i$，`cnt[i]++`。
3.  从$y$开始向上移动直到`lca(x,y)`，沿途点$i$，`cnt[i]++`。
4.  单次操作$O(n+\log n)$。
5.  总时间复杂度$O(kn+n\log n)$。

#### 优化做法

1.  对于每条路劲$x\to y$，利用树上差分，`diff[x]++,diff[y]++,diff[lca(x,y)]--,diff[fa[lca(x,y)]];`。
2.  单次修改$O(\log n)$，$k$次修改$O(k\log n)$。
3.  修改完毕后，跑一遍关于`diff[]`的树形DP，得到`cnt[i]`。
4.  $\max\limits_{i=1}^{n}cnt_i$即为答案。
5.  时间复杂度$O((n+k)logn)$。

#### 优化做法代码

```cpp
#include<bits/stdc++.h>
using namespace std;

const int N=1e5+5;
int n,k,diff[N],a[N],fa[N],dp[N][25],dep[N]; 
vector<int>nbr[N];

void pre_lca(int cur,int ffa)
{
	dep[cur]=dep[ffa]+1;
	fa[cur]=ffa;
	dp[cur][0]=ffa;
	for(int i=1;(1<<i)<=dep[cur];i++)
		dp[cur][i]=dp[dp[cur][i-1]][i-1];
	for(int i=0;i<nbr[cur].size();i++)
	{
		int nxt=nbr[cur][i];
		if(nxt==ffa)
			continue;
		pre_lca(nxt,cur);
	}
	return ;
}

int lca(int x,int y)
{
	if(dep[x]>dep[y])
		swap(x,y);
	for(int i=20;i>=0;i--)
		if(dep[dp[y][i]]>=dep[x])
			y=dp[y][i];
	if(x==y)
		return x;
	for(int i=20;i>=0;i--)
		if(dp[x][i]!=dp[y][i])
			x=dp[x][i],y=dp[y][i];
	return dp[x][0];
}

void dfs(int cur)
{
	a[cur]=diff[cur];
	for(int i=0;i<nbr[cur].size();i++)
	{
		int nxt=nbr[cur][i];
		if(nxt==fa[cur])
			continue;
		dfs(nxt);
		a[cur]+=a[nxt];
	}
	return ;
}

int main()
{
	cin>>n>>k;
	for(int i=1;i<=n-1;i++)
	{
		int u,v;
		cin>>u>>v;
		nbr[u].push_back(v);
		nbr[v].push_back(u);
	}
	pre_lca(1,0);
	while(k--)
	{
		int x,y;
		cin>>x>>y;
		int LCA=lca(x,y);
		diff[x]++;
		diff[y]++;
		diff[LCA]--;
		diff[fa[LCA]]--;
	}
	dfs(1);
	int maxi=-1e9;
	for(int i=1;i<=n;i++)
		maxi=max(maxi,a[i]);
	cout<<maxi;
	return 0;
}
```

### P3258 [JLOI2014] 松鼠的新家

#### 分析

1.  树上任意两点之间的简单路径是唯一的简单路径是唯一的。
2.  从`a[i]`到店`a[i+1]`的路径上每个点都要放$1$颗糖果，转化为`a[i]`到`a[i+1]`的树上差分，`diff[a[i]]++,diff[a[i+1]]++,diff[lca(a[i],a[i+1])]--,diff[fa[lca(a[i],a[i+1])]]`。
3.  一条路径的终点是下一条路径的起点，因此有重复，可以将`cnt[a[2]]`到`cnt[a[n]]`最后减$1$。

#### 代码

```cpp
#include<bits/stdc++.h>
using namespace std;

const int N=3e5+5;
int n,diff[N],a[N],z[N],fa[N],dp[N][25],dep[N]; 
vector<int>nbr[N];

void pre_lca(int cur,int ffa)
{
	dep[cur]=dep[ffa]+1;
	fa[cur]=ffa;
	dp[cur][0]=ffa;
	for(int i=1;(1<<i)<=dep[cur];i++)
		dp[cur][i]=dp[dp[cur][i-1]][i-1];
	for(int i=0;i<nbr[cur].size();i++)
	{
		int nxt=nbr[cur][i];
		if(nxt==ffa)
			continue;
		pre_lca(nxt,cur);
	}
	return ;
}

int lca(int x,int y)
{
	if(dep[x]>dep[y])
		swap(x,y);
	for(int i=21;i>=0;i--)
		if(dep[dp[y][i]]>=dep[x])
			y=dp[y][i];
	if(x==y)
		return x;
	for(int i=21;i>=0;i--)
		if(dp[x][i]!=dp[y][i])
			x=dp[x][i],y=dp[y][i];
	return dp[x][0];
}

void dfs(int cur)
{
	a[cur]=diff[cur];
	for(int i=0;i<nbr[cur].size();i++)
	{
		int nxt=nbr[cur][i];
		if(nxt==fa[cur])
			continue;
		dfs(nxt);
		a[cur]+=a[nxt];
	}
	return ;
}

int main()
{
	cin>>n;
	for(int i=1;i<=n;i++)
		cin>>z[i];
	for(int i=1;i<=n-1;i++)
	{
		int u,v;
		cin>>u>>v;
		nbr[u].push_back(v);
		nbr[v].push_back(u);
	}
	pre_lca(1,0);
	for(int i=2;i<=n;i++)
	{
		int x,y;
		x=z[i-1],y=z[i];
		int LCA=lca(x,y);
		diff[x]++;
		diff[y]++;
		diff[LCA]--;
		diff[fa[LCA]]--;
	}
	dfs(1);
	for(int i=2;i<=n;i++)
		a[z[i]]--;
	for(int i=1;i<=n;i++)
		cout<<a[i]<<"\n";
	return 0;
}
```

## 关于边权的树上差分

若修改树上$x\to y$路径上的边权均加一个val，等价于`diff[x]+=val,diff[y]+=val,diff[lca(x,y)]-=2*val`。

## 例题

### U143800 暗之连锁

#### 分析

1.  每一条附加边都会使树形成恰好一个环。
2.  每条附加边$(x,y)$会使得树上$x$到$y$的路径上的主要边被环盖$1$次。
3.  统计答案

1.  若$i$被覆盖的次数`cnt[i]==0`，则第一刀已经使得图不连通，第二刀在$m$条附加边中任选一条，`ans+=m`。
2.  若$i$被覆盖的次数`cnt[i]==1`，则第二刀砍环内的附加边，`ans++`。
3.  若$i$被覆盖的次数`cnt[i]>1`，则两刀无法砍断，无解。

4.  问题转化为如何快速求出主要边`cnt`，只需枚举$m$条附加边$i$，则$i$的两个端点$x$和$y$在树上的路径的主要边均被覆盖一次，差分处理即可。

#### 代码

```cpp
#include<bits/stdc++.h>
using namespace std;

const int N=3e5+5;
int n,k,diff[N],a[N],fa[N],dp[N][25],dep[N]; 
vector<int>nbr[N];

void pre_lca(int cur,int ffa)
{
	dep[cur]=dep[ffa]+1;
	fa[cur]=ffa;
	dp[cur][0]=ffa;
	for(int i=1;(1<<i)<=dep[cur];i++)
		dp[cur][i]=dp[dp[cur][i-1]][i-1];
	for(int i=0;i<nbr[cur].size();i++)
	{
		int nxt=nbr[cur][i];
		if(nxt==ffa)
			continue;
		pre_lca(nxt,cur);
	}
	return ;
}

int lca(int x,int y)
{
	if(dep[x]>dep[y])
		swap(x,y);
	for(int i=21;i>=0;i--)
		if(dep[dp[y][i]]>=dep[x])
			y=dp[y][i];
	if(x==y)
		return x;
	for(int i=21;i>=0;i--)
		if(dp[x][i]!=dp[y][i])
			x=dp[x][i],y=dp[y][i];
	return dp[x][0];
}

void dfs(int cur)
{
	a[cur]=diff[cur];
	for(int i=0;i<nbr[cur].size();i++)
	{
		int nxt=nbr[cur][i];
		if(nxt==fa[cur])
			continue;
		dfs(nxt);
		a[cur]+=a[nxt];
	}
	return ;
}

int main()
{
	cin>>n>>k;
	for(int i=1;i<=n-1;i++)
	{
		int u,v;
		cin>>u>>v;
		nbr[u].push_back(v);
		nbr[v].push_back(u);
	}
	pre_lca(1,0);
	for(int i=1;i<=k;i++)
	{
		int x,y;
		cin>>x>>y;
		int LCA=lca(x,y);
		diff[x]++;
		diff[y]++;
		diff[LCA]-=2;
	}
	dfs(1);
	int sum=0;
	for(int i=2;i<=n;i++)
	{
		if(a[i]==0)
			sum+=k; 
		if(a[i]==1)
			sum++;
	}
	cout<<sum;
	return 0;
}
```
